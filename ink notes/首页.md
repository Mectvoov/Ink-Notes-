function make(tag, className, text) {
  const el = document.createElement(tag);
  if (className) el.className = className;
  if (text !== undefined) el.textContent = text;
  return el;
}

function internalLink(path, label, className) {
  const link = make("a", `internal-link ${className || ""}`.trim(), label);
  link.href = path;
  link.dataset.href = path;
  return link;
}

function display(value, fallback = "未填写") {
  if (Array.isArray(value)) return value.length ? value.join(" · ") : fallback;
  if (value === null || value === undefined || value === "") return fallback;
  return String(value);
}

function cleanExcerpt(raw) {
  const text = raw
    .replace(/^---[\s\S]*?---\s*/, "")
    .replace(/<div class="mj-entry-meta">[\s\S]*?<\/div>/g, "")
    .replace(/%%[\s\S]*?%%/g, "")
    .replace(/!\[\[[^\]]+\]\]/g, "")
    .replace(/^#{1,6}\s+.*$/gm, "")
    .replace(/^>\s*\[![^\]]+\]\s*$/gm, "")
    .replace(/^>\s?/gm, "")
    .replace(/^-\s*$/gm, "")
    .replace(/^[-*_]{3,}\s*$/gm, "")
    .replace(/\s+/g, " ")
    .trim();
  return text.length > 30 ? `${text.slice(0, 29)}…` : text;
}

function extractFirstImage(raw) {
  const wikiMatch = raw.match(/!\[\[([^\]|]+)(?:\|[^\]]+)?\]\]/);
  if (wikiMatch) return wikiMatch[1].trim();
  const markdownMatch = raw.match(/!\[[^\]]*\]\(([^)]+)\)/);
  return markdownMatch ? markdownMatch[1].trim() : "";
}

function resolveImageFile(imagePath, sourcePath) {
  return app.metadataCache.getFirstLinkpathDest(imagePath, sourcePath)
    || app.vault.getAbstractFileByPath(imagePath)
    || app.vault.getFiles().find(file => file.path.endsWith(`/${imagePath}`) || file.name === imagePath);
}

function chineseMonth(day) {
  const year = String(day.year).split("").map(digit => chineseDigits[Number(digit)]).join("");
  const month = day.month <= 9
    ? chineseDigits[day.month]
    : `十${day.month === 10 ? "" : chineseDigits[day.month - 10]}`;
  return `${year}年 ${month}月`;
}

async function ensureCategoryHub(category) {
  const hubPath = categoryHubPath(category);
  const existing = app.vault.getAbstractFileByPath(hubPath);
  if (existing) return existing;
  return app.vault.create(hubPath, categoryHubText(category));
}

async function moveToCategory(page, category) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  if (!file) return;
  await ensureCategoryHub(category);
  await app.fileManager.processFrontMatter(file, frontmatter => {

    frontmatter.category_ref = `[[${categoryHubPath(category)}|${category}]]`;

}

function isValidCategoryName(name) {
  return name.length > 0 && name.length <= 8 && !/[\\/:*?"<>|#^\[\]]/.test(name);
}

async function persistCategories() {
  const ordered = categoryOptions.filter(category => category && category !== "未分类");
  ordered.push("未分类");
  if (categoryFile) {
    await app.fileManager.processFrontMatter(categoryFile, frontmatter => {
      frontmatter.categories = ordered;
    });
  }
  categoryOptions.splice(0, categoryOptions.length, ...ordered);
  for (const category of ordered) {
    await ensureCategoryHub(category);
  }
  await app.commands.executeCommandById("dataview:refresh-views");
}

async function addCategory(name) {
  const category = name.trim();
  if (!isValidCategoryName(category)) return false;
  if (categoryOptions.includes(category)) return false;
  const fallbackIndex = categoryOptions.indexOf("未分类");
  categoryOptions.splice(fallbackIndex >= 0 ? fallbackIndex : categoryOptions.length, 0, category);
  await persistCategories();
  return true;
}

async function renameCategory(oldName, newName) {
  const category = newName.trim();
  if (!isValidCategoryName(category) || oldName === "未分类" || categoryOptions.includes(category)) return false;

  const oldHub = app.vault.getAbstractFileByPath(categoryHubPath(oldName));
  if (oldHub) {
    await app.fileManager.renameFile(oldHub, categoryHubPath(category));
  }
  const newHub = app.vault.getAbstractFileByPath(categoryHubPath(category));
  if (newHub) {
    await app.vault.modify(newHub, categoryHubText(category));
  }

  for (const page of pages.filter(page => display(page.category) === oldName)) {
    await moveToCategory(page, category);
  }
  const index = categoryOptions.indexOf(oldName);
  if (index >= 0) categoryOptions[index] = category;
  await persistCategories();
  return true;
}

async function removeCategory(name) {
  if (name === "未分类") return false;
  const affected = pages.filter(page => display(page.category) === name);
  const message = affected.length
    ? `删除分类“${name}”？${affected.length} 篇日记将移到“未分类”。`
    : `删除分类“${name}”？`;
  if (!window.confirm(message)) return false;

  for (const page of affected) {
    await moveToCategory(page, "未分类");
  }
  const hub = app.vault.getAbstractFileByPath(categoryHubPath(name));
  if (hub) {
    await app.fileManager.trashFile(hub);
  }
  const index = categoryOptions.indexOf(name);
  if (index >= 0) categoryOptions.splice(index, 1);
  await persistCategories();
  return true;
}

async function deleteDiary(page) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  if (!file) return;
  if (!window.confirm(`删除“${page.file.name}”？`)) return;
  if (app.fileManager.trashFile) {
    await app.fileManager.trashFile(file);
  } else {
    await app.vault.delete(file, true);
  }
}

function showPageMenu(event, page) {
  event.preventDefault();
  event.stopPropagation();

    if (category === display(page.category)) item.classList.add("is-current");
    item.addEventListener("click", async clickEvent => {
      clickEvent.preventDefault();
      clickEvent.stopPropagation();
      closePageMenu();
      await moveToCategory(page, category);
    });
    menu.appendChild(item);
  }

  menu.appendChild(make("div", "mj-context-divider"));
  const remove = make("button", "mj-context-item is-danger", "删除");
  remove.type = "button";
  remove.addEventListener("click", async clickEvent => {
    clickEvent.preventDefault();
    clickEvent.stopPropagation();
    closePageMenu();
    await deleteDiary(page);
  });
  menu.appendChild(remove);

  document.body.appendChild(menu);
  menu.style.left = `${Math.min(event.clientX, window.innerWidth - menu.offsetWidth - 12)}px`;
  menu.style.top = `${Math.min(event.clientY, window.innerHeight - menu.offsetHeight - 12)}px`;
  activeMenu = menu;
  window.setTimeout(() => {
    document.addEventListener("click", closePageMenu, { once: true });
  }, 0);
}

function renderCategoryManager() {
  categoryManager.replaceChildren();
  const counts = new Map();
  pages.forEach(page => {
    const category = display(page.category);
    counts.set(category, (counts.get(category) || 0) + 1);
  });

  const list = make("div", "mj-category-manage-list");
  categoryOptions.forEach(category => {
    const item = make("div", "mj-category-manage-item");
    if (category === editingCategory) {
      item.classList.add("is-editing");
      const editForm = make("form", "mj-category-edit-form");
      const input = make("input", "mj-category-input");
      input.type = "text";
      input.maxLength = 8;
      input.value = category;
      input.setAttribute("aria-label", "新的分类名称");
      const save = make("button", "mj-small-action", "保存");
      save.type = "submit";
      const cancel = make("button", "mj-small-action", "取消");
      cancel.type = "button";
      cancel.addEventListener("click", () => {
        editingCategory = null;
        categoryError = "";
        renderCategoryManager();
      });
      editForm.addEventListener("submit", async event => {
        event.preventDefault();
        const ok = await renameCategory(category, input.value);
        if (!ok) {
          categoryError = "名称需为 1-8 个字，且不能与现有分类重复";
          renderCategoryManager();
          return;
        }
        editingCategory = null;
        categoryError = "";
      });
      editForm.appendChild(input);
      editForm.appendChild(save);
      editForm.appendChild(cancel);
      item.appendChild(editForm);
      list.appendChild(item);
      return;
    }

    const name = internalLink(categoryHubPath(category), category, "mj-category-manage-name");
    const count = make("span", "mj-category-manage-count", String(counts.get(category) || 0));
    item.appendChild(name);
    item.appendChild(count);

    if (category !== "未分类") {
      const actions = make("div", "mj-category-manage-actions");
      const rename = make("button", "mj-small-action", "改名");
      rename.type = "button";

        editingCategory = category;

      renderCategoryManager();
    }
  });

  categoryManager.appendChild(list);

}
