const pageKey = page => `${String(page.date).slice(0, 10)} ${String(page.time ?? "00:00")} ${page.file.name}`;
const pages = dv.pages('"ink notes/日记"')
  .where(p => p.type === "diary")
  .array()
  .filter(p => p.date)
  .sort((a, b) => pageKey(b).localeCompare(pageKey(a)));

const container = dv.container;
container.classList.add("mj-index-page");

function make(tag, className, text) {
  const el = document.createElement(tag);
  if (className) el.className = className;
  if (text !== undefined) el.textContent = text;
  return el;
}

function internalLink(path, label, className) {
  const link = make("a", `internal-link ${className || ""}`.trim(), label);
  link.href = path;
  link.dataset.href = path;
  return link;
}

function display(value, fallback = "未分类") {
  if (Array.isArray(value)) return value.length ? value.join(" · ") : fallback;
  if (value === null || value === undefined || value === "") return fallback;
  return String(value);
}

function cleanExcerpt(raw) {
  const text = raw
    .replace(/^---[\s\S]*?---\s*/, "")
    .replace(/<div class="mj-entry-meta">[\s\S]*?<\/div>/g, "")
    .replace(/%%[\s\S]*?%%/g, "")
    .replace(/!\[\[[^\]]+\]\]/g, "")
    .replace(/^#{1,6}\s+.*$/gm, "")
    .replace(/^>\s*\[![^\]]+\]\s*$/gm, "")
    .replace(/^>\s?/gm, "")
    .replace(/^-\s*$/gm, "")
    .replace(/^[-*_]{3,}\s*$/gm, "")
    .replace(/\s+/g, " ")
    .trim();
  return text.length > 30 ? `${text.slice(0, 29)}…` : text;
}

function extractFirstImage(raw) {
  const wikiMatch = raw.match(/!\[\[([^\]|]+)(?:\|[^\]]+)?\]\]/);
  if (wikiMatch) return wikiMatch[1].trim();
  const markdownMatch = raw.match(/!\[[^\]]*\]\(([^)]+)\)/);
  return markdownMatch ? markdownMatch[1].trim() : "";
}

function resolveImageFile(imagePath, sourcePath) {
  return app.metadataCache.getFirstLinkpathDest(imagePath, sourcePath)
    || app.vault.getAbstractFileByPath(imagePath)
    || app.vault.getFiles().find(file => file.path.endsWith(`/${imagePath}`) || file.name === imagePath);
}

const weekdayNames = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"];
const chineseDigits = ["零", "一", "二", "三", "四", "五", "六", "七", "八", "九"];

function chineseMonth(day) {
  const year = String(day.year).split("").map(digit => chineseDigits[Number(digit)]).join("");
  const month = day.month <= 9
    ? chineseDigits[day.month]
    : `十${day.month === 10 ? "" : chineseDigits[day.month - 10]}`;
  return `${year}年 ${month}月`;
}

const categoryFile = app.vault.getAbstractFileByPath("ink notes/分类配置.md");
const configuredCategories = app.metadataCache.getFileCache(categoryFile)?.frontmatter?.categories;
const baseCategories = Array.isArray(configuredCategories) && configuredCategories.length
  ? configuredCategories
  : ["生活", "工作", "学习", "灵感", "未分类"];
const categoryOptions = Array.from(new Set([
  ...baseCategories,
  ...pages.map(page => display(page.category)),
])).filter(Boolean);

let activeMenu = null;

function closePageMenu() {
  if (activeMenu) {
    activeMenu.remove();
    activeMenu = null;
  }
}

async function moveToCategory(page, category) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  if (!file) return;
  await app.fileManager.processFrontMatter(file, frontmatter => {
    frontmatter.category = category;
    frontmatter.category_ref = `[[ink notes/分类/${category}|${category}]]`;
  });
}

async function deleteDiary(page) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  if (!file) return;
  if (!window.confirm(`删除“${page.file.name}”？`)) return;
  if (app.fileManager.trashFile) {
    await app.fileManager.trashFile(file);
  } else {
    await app.vault.delete(file, true);
  }
}

function showPageMenu(event, page) {
  event.preventDefault();
  event.stopPropagation();
  closePageMenu();

  const menu = make("div", "mj-context-menu");
  menu.setAttribute("role", "menu");
  menu.appendChild(make("div", "mj-context-heading", "移动到分类"));

  for (const category of categoryOptions) {
    const item = make("button", "mj-context-item", category);
    item.type = "button";
    if (category === display(page.category)) item.classList.add("is-current");
    item.addEventListener("click", async clickEvent => {
      clickEvent.preventDefault();
      clickEvent.stopPropagation();
      closePageMenu();
      await moveToCategory(page, category);
    });
    menu.appendChild(item);
  }

  menu.appendChild(make("div", "mj-context-divider"));
  const remove = make("button", "mj-context-item is-danger", "删除");
  remove.type = "button";
  remove.addEventListener("click", async clickEvent => {
    clickEvent.preventDefault();
    clickEvent.stopPropagation();
    closePageMenu();
    await deleteDiary(page);
  });
  menu.appendChild(remove);

  document.body.appendChild(menu);
  menu.style.left = `${Math.min(event.clientX, window.innerWidth - menu.offsetWidth - 12)}px`;
  menu.style.top = `${Math.min(event.clientY, window.innerHeight - menu.offsetHeight - 12)}px`;
  activeMenu = menu;
  window.setTimeout(() => {
    document.addEventListener("click", closePageMenu, { once: true });
  }, 0);
}

if (!pages.length) {
  container.appendChild(make("div", "mj-index-empty", "还没有日记"));
} else {
  let currentMonth = "";

  for (const page of pages) {
    const day = dv.date(String(page.date).slice(0, 10));
    if (!day) continue;

    const month = chineseMonth(day);
    if (month !== currentMonth) {
      currentMonth = month;
      container.appendChild(make("div", "mj-index-month", month));
    }

    const file = app.vault.getAbstractFileByPath(page.file.path);
    let raw = "";
    let excerpt = "";
    if (file) {
      try {
        raw = await app.vault.cachedRead(file);
        excerpt = cleanExcerpt(raw);
      } catch (_) {
        excerpt = "";
      }
    }

    const row = make("div", "mj-index-entry");
    const dateBlock = make("div", "mj-index-date");
    dateBlock.appendChild(make("strong", "", day.toFormat("dd")));
    dateBlock.appendChild(make("span", "mj-index-time", display(page.time, "--:--:--")));
    dateBlock.appendChild(make("span", "mj-index-weekday", weekdayNames[day.weekday - 1]));

    const copy = make("div", "mj-index-copy");
    copy.appendChild(internalLink(page.file.path, excerpt || "空白日记", "mj-index-link"));

    const meta = make("div", "mj-index-meta");
    meta.appendChild(make("span", "", display(page.category)));
    if (page.mood) meta.appendChild(make("span", "", String(page.mood)));
    if (page.weather) meta.appendChild(make("span", "", String(page.weather)));
    copy.appendChild(meta);

    row.appendChild(dateBlock);
    row.appendChild(copy);
    const imagePath = extractFirstImage(raw);
    const imageFile = imagePath ? resolveImageFile(imagePath, page.file.path) : null;
    if (imageFile) {
      const thumbnail = make("img", "mj-index-thumb");
      thumbnail.src = app.vault.getResourcePath(imageFile);
      thumbnail.alt = "";
      row.classList.add("has-thumb");
      row.appendChild(thumbnail);
    }
    row.title = "右键：移动分类或删除";
    row.addEventListener("contextmenu", event => showPageMenu(event, page));
    container.appendChild(row);
  }
}
